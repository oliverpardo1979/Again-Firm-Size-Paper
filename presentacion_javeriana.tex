\begin{frame}{The Question}
  \idea{The paper asks whether Colombian workers earn more in larger firms, and what remains of that gap after controlling for worker characteristics.}

  \vspace{0.35cm}
  \begin{itemize}
    \item How much higher is real hourly pay in large firms relative to solo work and micro-firms?
    \item How much remains after controlling for worker characteristics, occupation, and fixed effects?
    \item Did the gap widen or narrow between 2008 and 2025?
    \item Does the size-wage relationship differ between formal and informal employment?
  \end{itemize}
\end{frame}

\begin{frame}{Main Results}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.49\textwidth}
      %\idea{Empirical fact}
      \vspace{0.25cm}
      \begin{itemize}
        \item Large firms pay more than solo work in the raw data.
        \item The relationship survives controls for observable characteristics.
        \item The 101+ premium vs solo workers is 37.1\% among informal and 16.5\% among formal workers.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      %\idea{Caution}
      \vspace{0.25cm}
      \begin{itemize}
        \item The wage-size premium is steeper in the informal sector versus the formal sector
        \item The premium does not prove causality or misallocation by itself.
        %\item The significance of the formal 101+/solo contrast depends on the inference procedure.
        %\item The salaried-only comparison qualifies the full-sample contrast between formal and informal workers.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\sectionframe{1. Motivation and Data}

\begin{frame}{Why Firm Size Matters}
  \idea{Firm size captures a visible dimension of the productive structure.}

  \vspace{0.35cm}
  \begin{itemize}
    \item In Colombia, a large share of employment is concentrated in own-account work and micro-firms.
    \item If hourly pay is systematically higher in larger firms, the firm-size distribution matters for wage inequality.
    \item The relationship can also be informative about frictions: barriers to growth, formalization, market access, or worker reallocation toward better firms.
  \end{itemize}
\end{frame}

\begin{frame}{The Central Caution}
  \idea{A conditional wage premium is not a causal effect of firm size.}

  \vspace{0.35cm}
  \begin{itemize}
    \item The conditional premium means that observationally similar workers earn more in larger firms.
    \item It does not prove that moving the same worker to a larger firm would raise aggregate output.
    \item The remaining gap may reflect unobserved worker characteristics or differences in employers' pay practices.
  \end{itemize}
  \source{Brown and Medoff (1989), Oi and Idson (1999), and the paper's discussion.}
\end{frame}

\begin{frame}{Data}
  \idea{We use GEIH because it covers the broader labor market, not only formal firms.}

  \vspace{0.3cm}
  \begin{itemize}
    \item Source: Gran Encuesta Integrada de Hogares, DANE.
    \item Period: 2008--2019 and 2021--2025.
    \item We exclude 2020 because of pandemic-related data limitations.
    \item Sample: 5.35 million worker observations pooled across repeated cross-sections, not a worker panel.
    \item Regression sample: 5.27 million observations with complete controls.
  \end{itemize}
\end{frame}

\begin{frame}{Main Variables}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \idea{Hourly pay}
      \vspace{0.25cm}
      \[
        w_{it} =
        \frac{12 \times MonthlyIncome_{it}}
        {52 \times WeeklyHours_{it}}
        \times
        \frac{CPI_{2025}}{CPI_t}
      \]
      \vspace{0.1cm}
      {\small Real hourly labor income in constant 2025 Colombian pesos.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \idea{Firm size}
      \vspace{0.25cm}
      \begin{itemize}
        \item Solo
        \item 2--3, 4--5, 6--10
        \item 11--19, 20--30, 31--50, 51--100
        \item 101+
      \end{itemize}
      {\small We combine 101--200 and 201+ into 101+ to match earlier questionnaires.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Firm Size and Occupational Position}
  \idea{Solo work describes the size of the reported unit, not the worker's occupational position.}
  \vspace{0.3cm}
  \begin{itemize}
    \item A household's sole domestic employee is a solo worker, but not an own-account worker.
    \item Own-account workers may report larger units when working with partners, unpaid family helpers, or as contractors.
    \item GEIH assigns state contractors to 101+ despite their own-account status.
    \item Occupational position is recorded separately and included as a control.
  \end{itemize}
  \source{Paper Section 2. DANE's GEIH collection manual (2025), pp. 89--90 and 115.}
\end{frame}

\begin{frame}{Formality}
  \idea{We define formality through pension contribution.}

  \vspace{0.35cm}
  \begin{itemize}
    \item A worker is formal if the worker contributes to pensions.
    \item A worker is informal if the worker does not contribute to pensions.
    \item Employed pension recipients are excluded from the formal-informal comparisons.
    \item This is a worker-level measure, not a measure of the firm's legal registration.
    \item The definition does not use firm size.
  \end{itemize}
\end{frame}

\begin{frame}{How to Read the U-Shape among Formal Workers}
  \idea{The U-shape among formal workers should not be read as evidence that all formal solo workers are high-earning professionals.}

  \vspace{0.35cm}
  \begin{itemize}
    \item In 2025, 82.1\% of formal solo workers were own-account workers.
    \item 50.9\% had higher education.
    \item 16.2\% were domestic workers.
    \item Worker composition is a plausible explanation for their high mean income.
  \end{itemize}
  \source{Paper Table~\paperref{tab:formal-worker-profile-2025}.}
\end{frame}
